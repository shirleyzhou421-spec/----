# 灯下伴读 App · 研发交付与源码归档文档（V1.6）

* **文档用途：** 研发团队代码评审、工程拆分与接口联调交付归档
* **建议目录结构：**
  * `index.html`（页面主结构与语义化组件容器）
  * `styles/main.css`（深夜视觉变量、响应式排版、2:3 书封与双模式阅读样式）
  * `scripts/app.js`（15 本公版书库配置、双模式切章、Web Audio 20 段声景合成、助眠朗读与本地状态持久化）

---

## 一、 交付架构与约定说明

1. **三层解耦拆分：** 本归档将单文件原型拆分为标准的 `HTML`、`CSS`、`JS` 三个代码块，研发团队可直接复制落地为标准静态工程或接入构建工具。
2. **豆包语音与离线兜底约定：**
   * 助眠朗读与「问周先生」语音模式在前端预留了语音调用衔接位；当未连接服务端语音中转或网络不可用时，自动平滑回退至设备浏览器自带中文人声（`speechSynthesis`）及标注清晰的离线预设回应。
   * 严禁在前端代码中硬编码任何服务端 API Key，正式上线时统一通过同源后端接口转发。
3. **本地持久化键名：** 所有阅读进度（`bookId`、`chapterIndex`、`paraIndex`、`charOffset`）、偏好设置、段落随感、夜话历史与夜读卡片统一存储在 `localStorage["dengxia_app_state_v16"]`。

function renderCurrentChapter() {
  const book = getCurrentBook();
  if (appState.chapterIndex >= book.chapters.length) appState.chapterIndex = 0;
  const chapter = book.chapters[appState.chapterIndex];

  document.getElementById("reader-book-name").textContent = book.title;
  document.getElementById("reader-chapter-meta").textContent = `${book.author} · 第 ${appState.chapterIndex + 1} / ${book.chapters.length} 篇`;
  document.getElementById("reader-chapter-title").textContent = chapter.title;

  const isFlip = appState.pageMode === "flip";
  document.getElementById("reader-scroll-area").classList.toggle("hidden", isFlip);
  document.getElementById("reader-flip-area").classList.toggle("hidden", !isFlip);

  if (!isFlip) {
    renderVerticalScrollChapter(book, chapter);
  } else {
    renderHorizontalFlipChapter(book, chapter);
  }
  updateSettingsButtonsUI();
  saveState();
}

function renderVerticalScrollChapter(book, chapter) {
  const container = document.getElementById("reader-paragraphs");
  container.innerHTML = "";
  chapter.paragraphs.forEach((text, idx) => {
    const p = document.createElement("p");
    p.textContent = text;
    p.dataset.pidx = idx;
    const hasNote = appState.reflections.some(
      r => r.bookId === book.id && r.chapterIndex === appState.chapterIndex && r.paraIndex === idx
    );
    if (hasNote) p.classList.add("has-note");
    if (isReadingAloud && readingParaIndex === idx) p.classList.add("reading-highlight");
    p.addEventListener("click", () => {
      appState.paraIndex = idx;
      appState.charOffset = 0;
      saveState();
    });
    container.appendChild(p);
  });

  const isFirst = appState.chapterIndex === 0;
  const isLast = appState.chapterIndex === book.chapters.length - 1;
  document.getElementById("btn-bottom-prev").style.opacity = isFirst ? "0.35" : "1";
  document.getElementById("chapter-end-nav").classList.toggle("hidden", isLast);
  document.getElementById("book-finished-box").classList.toggle("hidden", !isLast);

  const scrollArea = document.getElementById("reader-scroll-area");
  if (appState.paraIndex > 0 && container.children[appState.paraIndex]) {
    setTimeout(() => {
      container.children[appState.paraIndex].scrollIntoView({ behavior: "smooth", block: "center" });
    }, 60);
  } else {
    scrollArea.scrollTop = 0;
  }
}

function renderHorizontalFlipChapter(book, chapter) {
  document.getElementById("flip-chapter-label").textContent = `${book.title} · ${chapter.title}`;
  const viewportH = window.innerHeight - 175;
  const linesPerScreen = Math.max(4, Math.floor(viewportH / (appState.fontSize * 1.95)));
  const charsPerLine = Math.max(14, Math.floor(Math.min(window.innerWidth - 64, 620) / appState.fontSize));
  const charsPerPage = Math.max(90, Math.floor(linesPerScreen * charsPerLine * 0.78));

  paginatedPages = [];
  let currentBucket = [];
  let currentChars = 0;
  let startPara = 0;

  chapter.paragraphs.forEach((para, idx) => {
    if (currentChars + para.length > charsPerPage && currentBucket.length > 0) {
      paginatedPages.push({ paras: currentBucket, startPara });
      currentBucket = [];
      currentChars = 0;
      startPara = idx;
    }
    currentBucket.push({ text: para, idx });
    currentChars += para.length;
  });
  if (currentBucket.length > 0) paginatedPages.push({ paras: currentBucket, startPara });

  const matchedPage = paginatedPages.findIndex(
    (pg, i) => appState.paraIndex >= pg.startPara && (i === paginatedPages.length - 1 || appState.paraIndex < paginatedPages[i + 1].startPara)
  );
  if (matchedPage >= 0) appState.flipPageIndex = matchedPage;
  if (appState.flipPageIndex >= paginatedPages.length) appState.flipPageIndex = 0;
  renderFlipPageContent();
}

function renderFlipPageContent() {
  const pageObj = paginatedPages[appState.flipPageIndex] || { paras: [], startPara: 0 };
  document.getElementById("flip-page-indicator").textContent = `第 ${appState.flipPageIndex + 1} / ${paginatedPages.length} 页`;
  const container = document.getElementById("flip-page-content");
  container.innerHTML = "";
  pageObj.paras.forEach(item => {
    const p = document.createElement("p");
    p.textContent = item.text;
    p.dataset.pidx = item.idx;
    if (isReadingAloud && readingParaIndex === item.idx) p.classList.add("reading-highlight");
    container.appendChild(p);
  });
  appState.paraIndex = pageObj.startPara;
  saveState();
}

function turnPage(delta) {
  if (isTextSelectedLocked) return;
  const next = appState.flipPageIndex + delta;
  if (next >= 0 && next < paginatedPages.length) {
    appState.flipPageIndex = next;
    renderFlipPageContent();
  } else if (next >= paginatedPages.length) {
    nextChapter();
  } else if (next < 0) {
    prevChapter();
  }
}

function prevChapter() {
  if (appState.chapterIndex > 0) {
    appState.chapterIndex--;
    appState.paraIndex = 0;
    appState.charOffset = 0;
    appState.flipPageIndex = 0;
    if (isReadingAloud) readingParaIndex = 0;
    renderCurrentChapter();
  } else {
    showToast("已是第一篇");
  }
}

function nextChapter() {
  const book = getCurrentBook();
  if (appState.chapterIndex < book.chapters.length - 1) {
    appState.chapterIndex++;
    appState.paraIndex = 0;
    appState.charOffset = 0;
    appState.flipPageIndex = 0;
    if (isReadingAloud) readingParaIndex = 0;
    renderCurrentChapter();
  } else {
    showToast("本章已读完，可合书熄灯或返回书架");
  }
}

document.addEventListener("keydown", (e) => {
  if (currentTab === "companion" && appState.pageMode === "flip" && !isTextSelectedLocked) {
    if (e.key === "ArrowRight") turnPage(1);
    if (e.key === "ArrowLeft") turnPage(-1);
  }
});

let touchStartX = 0;
document.getElementById("reader-flip-area").addEventListener("touchstart", (e) => {
  touchStartX = e.changedTouches[0].clientX;
}, { passive: true });
document.getElementById("reader-flip-area").addEventListener("touchend", (e) => {
  if (isTextSelectedLocked) return;
  const diff = e.changedTouches[0].clientX - touchStartX;
  if (Math.abs(diff) > 45) turnPage(diff < 0 ? 1 : -1);
}, { passive: true });

let isTextSelectedLocked = false;

document.addEventListener("selectionchange", () => {
  if (currentTab !== "companion") return;
  const sel = window.getSelection();
  const text = sel ? sel.toString().trim() : "";
  const toolbar = document.getElementById("selection-toolbar");
  const zL = document.getElementById("flip-zone-left");
  const zR = document.getElementById("flip-zone-right");

  if (text.length > 0 && !document.getElementById("companion-reader").classList.contains("hidden")) {
    selectedQuote = text;
    isTextSelectedLocked = true;
    zL.classList.add("hidden");
    zR.classList.add("hidden");
    if (isReadingAloud) pauseReadAloudTemporarily();

    const anchorNode = sel.anchorNode;
    const pEl = anchorNode && (anchorNode.nodeType === 1 ? anchorNode.closest("p") : anchorNode.parentElement?.closest("p"));
    if (pEl && pEl.dataset.pidx !== undefined) {
      selectedParaIdx = Number(pEl.dataset.pidx);
      appState.paraIndex = selectedParaIdx;
      appState.charOffset = pEl.textContent.indexOf(text);
      saveState();
    }
    toolbar.classList.remove("hidden");
  } else {
    isTextSelectedLocked = false;
    toolbar.classList.add("hidden");
    zL.classList.remove("hidden");
    zR.classList.remove("hidden");
  }
});

let isReadingAloud = false;

let sleepTimerHandle = null;
let fallbackSimTimer = null;

function speakCurrentParagraph() {
  if (!isReadingAloud) return;
  const book = getCurrentBook();
  const chapter = book.chapters[appState.chapterIndex];
  if (!chapter || readingParaIndex >= chapter.paragraphs.length) {
    if (appState.sleepTimerMin === -1) {
      stopReadAloud();
      stopSound();
      showToast("已读完本章，助眠朗读与轻音已轻柔停止");
      return;
    }
    if (appState.chapterIndex < book.chapters.length - 1) {
      appState.chapterIndex++;
      appState.paraIndex = 0;
      readingParaIndex = 0;
      renderCurrentChapter();
      speakCurrentParagraph();
    } else {
      stopReadAloud();
      showToast("全书已朗读完毕");
    }
    return;
  }

  appState.paraIndex = readingParaIndex;
  saveState();
  document.querySelectorAll(".reader-prose p").forEach(p => p.classList.remove("reading-highlight"));

  if (appState.pageMode === "scroll") {
    const pEl = document.querySelector(`#reader-paragraphs p[data-pidx="${readingParaIndex}"]`);
    if (pEl) {
      pEl.classList.add("reading-highlight");
      pEl.scrollIntoView({ behavior: "smooth", block: "center" });
    }
  } else {
    const targetPage = paginatedPages.findIndex(
      (pg, i) => readingParaIndex >= pg.startPara && (i === paginatedPages.length - 1 || readingParaIndex < paginatedPages[i + 1].startPara)
    );
    if (targetPage >= 0 && targetPage !== appState.flipPageIndex) {
      appState.flipPageIndex = targetPage;
      renderFlipPageContent();
    }
    const pFlip = document.querySelector(`#flip-page-content p[data-pidx="${readingParaIndex}"]`);
    if (pFlip) pFlip.classList.add("reading-highlight");
  }

  const text = chapter.paragraphs[readingParaIndex];
  // 预留豆包语音接口替换位；本地静态模式回退设备中文人声
  if ("speechSynthesis" in window) {
    window.speechSynthesis.cancel();
    const utter = new SpeechSynthesisUtterance(text);
    utter.lang = "zh-CN";
    utter.rate = appState.readRate;
    utter.pitch = 0.92;
    utter.onend = () => {
      if (isReadingAloud) {
        readingParaIndex++;
        speakCurrentParagraph();
      }
    };
    utter.onerror = () => simulateParagraphAdvance(text);
    window.speechSynthesis.speak(utter);
  } else {
    simulateParagraphAdvance(text);
  }
}

function simulateParagraphAdvance(text) {
  clearTimeout(fallbackSimTimer);
  const durationMs = Math.max(3000, text.length * 190 / appState.readRate);
  fallbackSimTimer = setTimeout(() => {
    if (isReadingAloud) {
      readingParaIndex++;
      speakCurrentParagraph();
    }
  }, durationMs);
}

function setSleepTimer(min) {
  appState.sleepTimerMin = min;
  saveState();
  clearTimeout(sleepTimerHandle);
  if (min > 0) {
    sleepTimerHandle = setTimeout(() => {
      stopReadAloud();
      stopSound();
      appState.sleepTimerMin = 0;
      updateSettingsButtonsUI();
      showToast(`${min} 分钟睡眠定时已到，朗读与轻音已停止`);
    }, min * 60 * 1000);
    showToast(`已设定 ${min} 分钟后停止朗读与轻音`);
  } else if (min === -1) {
    showToast("已设定读完本章后停止");
  } else {
    showToast("已关闭睡眠定时");
  }
  updateSettingsButtonsUI();
}

function handleLocalTxtImport(e) {
  const file = e.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(evt) {
    const rawText = evt.target.result || "";
    const lines = rawText.split(/\r?\n/).map(s => s.trim()).filter(Boolean);
    if (lines.length === 0) return showToast("文件内容为空");
    const chapters = [];
    for (let i = 0; i < lines.length; i += 12) {
      chapters.push({ title: `第 ${chapters.length + 1} 节`, paragraphs: lines.slice(i, i + 12) });
    }
    const cleanName = file.name.replace(/\.txt$/i, "");
    appState.customBooks.unshift({
      id: "local_" + Date.now(),
      title: cleanName,
      author: "本地导入",
      category: "散文随笔",
      defaultMode: "闲谈",
      sourceInfo: `用户于本地设备私密导入的文本《${file.name}》。`,
      chapters
    });
    saveState();
    renderBookshelf();
    showToast(`已导入《${cleanName}》`);
  };
  reader.readAsText(file, "UTF-8");
}

function playSynthesizedSound(soundObj) {
  if (appState.soundPlaying && appState.soundId === soundObj.id) return stopSound();
  stopAudioNodesOnly();

  const AudioContextClass = window.AudioContext || window.webkitAudioContext;
  if (!audioCtx) audioCtx = new AudioContextClass();
  if (audioCtx.state === "suspended") audioCtx.resume();

  const masterGain = audioCtx.createGain();
  const effectiveVol = appState.soundVolume * (isReadingAloud ? 0.32 : 1.0);
  masterGain.gain.setValueAtTime(effectiveVol * 0.25, audioCtx.currentTime);
  masterGain.connect(audioCtx.destination);

  if (soundObj.type === "sine") {
    const osc = audioCtx.createOscillator();
    osc.type = "sine";
    osc.frequency.value = soundObj.freq;
    const lfo = audioCtx.createOscillator();
    lfo.frequency.value = soundObj.lfo;
    const lfoGain = audioCtx.createGain();
    lfoGain.gain.value = 0.08;
    lfo.connect(lfoGain);
    lfoGain.connect(masterGain.gain);
    osc.connect(masterGain);
    osc.start();
    lfo.start();
    activeSoundNodes = { osc, lfo, masterGain };
  } else {
    const bufferSize = 2 * audioCtx.sampleRate;
    const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
    const output = noiseBuffer.getChannelData(0);
    let b0 = 0, b1 = 0, b2 = 0, lastOut = 0;
    for (let i = 0; i < bufferSize; i++) {
      const white = Math.random() * 2 - 1;
      if (soundObj.type === "pink") {
        b0 = 0.99765 * b0 + white * 0.099046;
        b1 = 0.96300 * b1 + white * 0.2965164;
        b2 = 0.57000 * b2 + white * 1.0526913;
        output[i] = (b0 + b1 + b2 + white * 0.1848) * 0.12;
      } else {
        output[i] = (lastOut + 0.02 * white) / 1.02;
        lastOut = output[i];
        output[i] *= 0.85;
      }
    }
    const source = audioCtx.createBufferSource();
    source.buffer = noiseBuffer;
    source.loop = true;
    const filter = audioCtx.createBiquadFilter();
    filter.type = "lowpass";
    filter.frequency.value = soundObj.freq;
    source.connect(filter);
    filter.connect(masterGain);
    source.start();
    activeSoundNodes = { source, masterGain };
  }

  appState.soundId = soundObj.id;
  appState.soundPlaying = true;
  document.getElementById("sound-master-status").textContent = `播放中 · ${soundObj.name}`;
  renderSoundsGrid();
}

function stopAudioNodesOnly() {
  if (activeSoundNodes) {
    try {
      if (activeSoundNodes.osc) activeSoundNodes.osc.stop();
      if (activeSoundNodes.lfo) activeSoundNodes.lfo.stop();
      if (activeSoundNodes.source) activeSoundNodes.source.stop();
    } catch (e) {}
    activeSoundNodes = null;
  }
}
